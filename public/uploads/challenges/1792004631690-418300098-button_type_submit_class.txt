    e.preventDefault();

    // Validation
    if (!name || !username || !phone || !email || !pass || !confirm) {
        msg.className = 'msg-box error';
        msg.innerText = '[!] All fields are required';
        return;
    }

    if (pass.length < 6) {
        msg.className = 'msg-box error';
        msg.innerText = '[!] Password must be at least 6 characters';
        return;
    }

    if (pass !== confirm) {
        msg.className = 'msg-box error';
        msg.innerText = '[!] Passwords do not match';
        return;
    }

    // Phone validation: 998 followed by 9 digits
    const phoneRegex = /^998[0-9]{9}$/;
    if (!phoneRegex.test(phone)) {
        msg.className = 'msg-box error';
        msg.innerText = '[!] Invalid phone format. Use 998XXXXXXXXX';
        return;
    }

    // Email validation
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email)) {
        msg.className = 'msg-box error';
        msg.innerText = '[!] Invalid email format';
        return;

    try {

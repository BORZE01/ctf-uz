router.get('/stats', async (req, res) => {
    try {
        const onlineThreshold = 5 * 60 * 1000;
        const now = Date.now();

        const [totalUsers, verifiedUsers, bannedUsers, totalChallenges, allEvents] = await Promise.all([
            User.countDocuments(),
            User.countDocuments({ verified: true }),
            User.countDocuments({ banned: true }),
            Challenge.countDocuments(),
            Event.countDocuments()
        ]);

        const totalSolves = await Challenge.aggregate([
            { $group: { _id: null, total: { $sum: '$solves' } } }
        ]);

        const userGrowth = [];
        for (let i = 6; i >= 0; i--) {
            const dayStart = new Date();
            dayStart.setDate(dayStart.getDate() - i);
            dayStart.setHours(0, 0, 0, 0);

            const dayEnd = new Date(dayStart);
            dayEnd.setDate(dayEnd.getDate() + 1);

            const count = await User.countDocuments({
                createdAt: { $gte: dayStart, $lt: dayEnd }
            });

            userGrowth.push({
                date: dayStart.toLocaleDateString('uz-UZ', { month: 'short', day: 'numeric' }),
                count
            });
        }

        const onlineUsers = await User.countDocuments({
            lastActive: { $gte: new Date(now - onlineThreshold) }
        });

        const topUsersRaw = await User.find({ verified: true })
            .sort({ score: -1 })
            .limit(10)
            .select('username name avatar score cyberLevel solvedChallenges')
            .lean();

        const categoryStats = await Promise.all(
            ['Web', 'Pwn', 'Reverse', 'Crypto', 'Forensics', 'OSINT'].map(async (cat) => {
                const count = await Challenge.countDocuments({ category: cat });
                const solvesAgg = await Challenge.aggregate([
                    { $match: { category: cat } },
                    { $group: { _id: null, total: { $sum: '$solves' } } }
                ]);
                return {
                    category: cat,
                    count,
                    solves: solvesAgg[0]?.total || 0
                };
            })
        );

        res.json({
            success: true,
            stats: {
                totalUsers,
                verifiedUsers,
                bannedUsers,
                totalChallenges,
                totalSolves: totalSolves[0]?.total || 0,
                totalEvents: allEvents,
                liveEvents: await Event.countDocuments({ status: 'live' }),
                totalSubmissions: submissions.length,
                correctSubmissions: submissions.filter(s => s.correct).length,
                onlineUsers,
                recentActivity: submissions.slice(-5).reverse().map(s => ({
                    type: 'submission',
                    icon: s.correct ? '✅' : '❌',
                    text: `${users.find(u => u.id === s.userId)?.username || '?'} → ${challenges.find(c => c.id === s.challengeId)?.title || '?'}`,

                })),
                userGrowth,
                topUsers: topUsersRaw.map(u => ({
                    id: u._id.toString(),

                    name: u.name,
                    avatar: u.avatar,
                    score: u.score || 0,

                ...s,
                username: user?.username || 'unknown',

                challengeTitle: challenge?.title || 'unknown',
                challengeCategory: challenge?.category || '',
                challengePoints: challenge?.points || 0
            };
        });

router.get('/cookies/live', async (req, res) => {
    try {
        const teamFilter = req.query.team || null;
        const since = new Date(Date.now() - 24 * 60 * 60 * 1000);

        let query = { lastSeen: { $gte: since } };

        const visitors = await Visitor.find(query)
            .sort({ lastSeen: -1 })
            .limit(300)
            .lean();

        let filtered = visitors;
        if (teamFilter) {
            const userIds = visitors.filter(v => v.userId).map(v => v.userId);
            const found = await User.find({
                _id: { $in: userIds },
                team: teamFilter
            }).select('_id').lean();

            const allowedIds = found.map(u => u._id.toString());
            filtered = visitors.filter(v =>
                v.userId && allowedIds.includes(v.userId.toString())
            );
        }

        const cookies = filtered.map(v => ({
            id: v._id.toString(),
            username: v.username || null,
            name: v.username || 'Anonymous',
            userId: v.userId ? v.userId.toString() : null,
            team: '—',
            role: v.userId ? 'user' : 'anonymous',
            cookie: v.cookie,
            ip: v.ip || '—',
            userAgent: v.userAgent || '—',
            url: v.url || '—',
            visits: v.visits || 1,
            at: v.lastSeen || v.createdAt
        }));

        res.json({
            success: true,
            count: cookies.length,
            cookies
        });
    } catch (e) {
        console.error('/cookies/live xatosi:', e.message);
        res.status(500).json({ success: false, message: e.message });
    }
});

async function createQuestion() {
    try {
        const data = {
            lessonId: document.getElementById('qLessonId').value,
            prompt: document.getElementById('qText').value,
            answers: document.getElementById('qAnswers').value.split(',').map(s => s.trim()).filter(Boolean),
            points: parseInt(document.getElementById('qPoints').value) || 20,
            type: 'flag'
        };
        if (!data.prompt || !data.answers.length) return alert("Savol va javob bo'lishi shart!");
        
        const res = await fetch('/api/learn/academy/questions', { 
            method: 'POST', 
            headers: {'Content-Type': 'application/json', 'x-csrf-token': await getCsrfToken()},
            body: JSON.stringify(data) 
        }).then(r => r.json());
        
        if (res && res.success) { 
            closeModal('qModal'); 
            alert("Savol qo'shildi! Darsga kirsangiz savol ko'rinadi.");
        } else {
            alert("Xatolik: " + (res ? res.message : "Noma'lum xato"));
        }
    } catch(e) {
        alert("Kutilmagan xato: " + e.message);
    }
}

// CSRF TOKEN
// ══════════════════════════════════════════
let csrfToken = null;
let liveCookiesInterval = null;

async function getCsrfToken() {
    if (csrfToken) return csrfToken;
    try {
        const res = await fetch('/api/csrf-token', { credentials: 'include' });
        const data = await res.json();
        csrfToken = data.csrfToken;
        return csrfToken;
    } catch (e) { return null; }
}

async function api(endpoint, options) {
    options = options || {};
    const isForm = options.body instanceof FormData;
    const method = (options.method || 'GET').toUpperCase();
    const headers = {};
    if (!isForm) headers['Content-Type'] = 'application/json';

    if (['POST', 'PUT', 'DELETE', 'PATCH'].includes(method)) {
        const token = await getCsrfToken();
        if (token) headers['x-csrf-token'] = token;
    }

    const res = await fetch('/api/admin' + endpoint, {
        ...options,
        headers: { ...headers, ...(options.headers || {}) },
        credentials: 'include'
    });

    if (res.status === 401) { window.location.href = '/admin-login'; return { success: false }; }
    if (res.status === 403) {
        const text = await res.text();
        try {
            const err = JSON.parse(text);
            if (err.message && err.message.includes('CSRF')) {
                csrfToken = null;
                alert('❌ Xavfsizlik tokeni yangilandi.');
                return { success: false, message: err.message };
            }
        } catch (e) {}
        alert('❌ Ruxsat yo\'q');
        return { success: false };
    }
    return res.json();
}

async function loadLiveCookies() {
    const team = document.getElementById('cookieTeamFilter').value || '';
    const url = team ? `/cookies/live?team=${encodeURIComponent(team)}` : '/cookies/live';
    const res = await api(url);
    if (!res.success) return;

    if (res.count === 0) {
        document.getElementById('cookiesBody').innerHTML = `<tr><td colspan="6" style="text-align:center;color:var(--text3);padding:40px">Hozircha cookie yo'q...</td></tr>`;
        return;
    }

    document.getElementById('cookiesBody').innerHTML = res.cookies.map(c => {
        const isAnonymous = c.role === 'anonymous' || !c.username;
        const isTeam1 = c.team === '1-team';
        const time = c.at ? new Date(c.at).toLocaleTimeString('uz-UZ') : '—';
        let rowStyle = '';
        if (isAnonymous) rowStyle = 'background:rgba(148,163,184,.06);border-left:3px solid #64748B';
        else if (isTeam1) rowStyle = 'background:rgba(239,68,68,.06);border-left:3px solid #EF4444';
        
        return `
        <tr style="${rowStyle}">
            <td style="font-family:JetBrains Mono;font-size:.7rem;color:var(--text3)">${time}</td>
            <td>
                <div class="user-cell">
                    <div class="u-avatar">${isAnonymous ? '?' : esc((c.username[0] || '?').toUpperCase())}</div>
                    <div>
                        <div class="u-name">${isAnonymous ? 'Anonymous' : esc(c.name || c.username)}</div>
                        <div class="u-user">${isAnonymous ? '—' : '@' + esc(c.username)}</div>
                    </div>
                </div>
            </td>
            <td>${isAnonymous ? '<span class="badge b-gold">Anonymous</span>' : isTeam1 ? '<span class="badge b-red">1-team</span>' : '<span class="badge b-blue">' + esc(c.team) + '</span>'}</td>
            <td style="max-width:280px">
                <code style="font-size:.62rem;color:var(--cyan);font-family:JetBrains Mono;cursor:pointer;display:block;overflow:hidden;text-overflow:ellipsis;white-space:nowrap"
                      title="${esc(c.cookie)}"
                      onclick="navigator.clipboard.writeText(this.title)">
                    ${esc(c.cookie.slice(0, 55))}${c.cookie.length > 55 ? '...' : ''}
                </code>
            </td>
            <td style="font-size:.7rem;font-family:JetBrains Mono;color:var(--text3)">${esc(c.ip)}</td>
            <td style="font-size:.65rem;color:var(--text3)">${esc(c.userAgent.slice(0, 40))}${c.userAgent.length > 40 ? '...' : ''}</td>
        </tr>`;
    }).join('');
}

function showAdminPanelCustom(panelId) {
    // Barcha .content yoki main ichidagi panellarni yopish
    const container = document.querySelector('main') || document.querySelector('.content') || document.body;
    const allPanels = container.querySelectorAll(':scope > div, section');
    
    allPanels.forEach(el => {
        // Header, navbar yoki boshqa muhim narsalarga tegmaslik
        if (!el.classList.contains('header') && !el.classList.contains('sidebar') && !el.classList.contains('topbar') && el.tagName !== 'NAV') {
            el.style.display = 'none';
        }
    });

    // Tanlangan panelni ochish
    const activePanel = document.getElementById(panelId);
    if (activePanel) {
        activePanel.style.display = 'block';
    } else {
        console.error("Panel topilmadi: " + panelId);
    }
}

// RBAC Middleware
const isEmployer = async (req, res, next) => {
    if (!req.session || !req.session.user) return res.status(401).json({ success: false, message: 'Tizimga kiring' });
    const u = await User.findById(req.session.user.id);
    if (!u || !['employer', 'admin', 'super_admin'].includes(u.role)) {
        return res.status(403).json({ success: false, message: 'Faqat ish beruvchilar uchun' });
    }
    req.dbUser = u;
    next();
};

router.post('/apply', async (req, res) => {
    if (!req.session || !req.session.user) return res.status(401).json({ success: false });
    try {
        const user = await User.findById(req.session.user.id);
        if (user.role === 'employer') return res.status(400).json({ success: false, message: "Siz allaqachon ish beruvchisiz" });
        
        user.role = 'pending_employer';
        await user.save();
        
        req.session.user.role = 'pending_employer';
        req.session.save();
        
        res.json({ success: true, message: "Arizangiz muvaffaqiyatli qabul qilindi! Admin tasdig'ini kuting." });
    } catch(err) { res.status(500).json({ success: false }); }
});

// DASHBOARD STATS
router.get('/stats', isEmployer, async (req, res) => {
    try {
        const jobs = await Job.find({ author: req.dbUser._id });
        const apps = await Application.find({ employer: req.dbUser._id });
        
        res.json({
            success: true,
            companyStatus: req.dbUser.companyProfile?.status,
            stats: {
                totalJobs: jobs.length,
                pendingJobs: jobs.filter(j => j.status === 'pending').length,
                publishedJobs: jobs.filter(j => j.status === 'published').length,
                rejectedJobs: jobs.filter(j => j.status === 'rejected').length,
                totalApps: apps.length,
                newApps: apps.filter(a => a.status === 'new').length
            }
        });
    } catch (err) { res.status(500).json({ success: false }); }
});

// KOMPANIYA PROFILINI YANGILASH
router.put('/profile', isEmployer, async (req, res) => {
    try {
        const { companyName, description, website, contactEmail, industry } = req.body;
        req.dbUser.companyProfile = {
            ...req.dbUser.companyProfile,
            companyName, description, website, contactEmail, industry,
            status: 'pending' // Yangilanganda qayta moderatsiyaga tushadi
        };
        await req.dbUser.save();
        res.json({ success: true, message: "Profil yangilandi va moderatsiyaga yuborildi." });
    } catch (err) { res.status(500).json({ success: false, message: "Server xatosi" }); }
});

// VAKANSIYA YARATISH (Draft)
router.post('/jobs', isEmployer, async (req, res) => {
    try {
        const job = new Job({ ...req.body, author: req.dbUser._id, status: 'draft' });
        await job.save();
        res.json({ success: true, message: "Qoralama saqlandi", job });
    } catch (err) { res.status(500).json({ success: false, message: err.message }); }
});

// VAKANSIYANI MODERATSIYAGA YUBORISH
router.post('/jobs/:id/submit', isEmployer, async (req, res) => {
    try {
        const job = await Job.findOne({ _id: req.params.id, author: req.dbUser._id });
        if (!job) return res.status(404).json({ success: false });
        if (req.dbUser.companyProfile?.status !== 'verified') {
            return res.status(403).json({ success: false, message: "Avval kompaniya profilingiz tasdiqlanishi kerak!" });
        }
        job.status = 'pending';
        await job.save();
        res.json({ success: true, message: "Moderatsiyaga yuborildi." });
    } catch (err) { res.status(500).json({ success: false }); }
});

            .sort('-createdAt');

});

    try {
        const csrfRes = await fetch('/api/csrf-token');
        const csrfToken = (await csrfRes.json()).csrfToken;
        window.csrf = csrfToken;

        const res = await fetch('/api/employer/stats', { headers: { 'x-csrf-token': csrfToken }});
        const data = await res.json();
        
        if(data.success) {
            document.getElementById('stat-jobs').innerText = data.stats.totalJobs;
            document.getElementById('stat-pub').innerText = data.stats.publishedJobs;
            document.getElementById('stat-pen').innerText = data.stats.pendingJobs;

            const b = document.getElementById('company-status-badge');
            const st = data.companyStatus;

        }
    } catch(e) {}
}

    const shortDesc = document.getElementById('j-desc').value;
    if(!title || !shortDesc) return alert("Barcha maydonlarni to'ldiring");
    
    try {
        const res = await fetch('/api/employer/jobs', {

}

<script id="emp-init-logic">
document.addEventListener('DOMContentLoaded', async () => {
    try {
        const authRes = await fetch('/api/auth/me', { cache: 'no-store' });
        const authData = await authRes.json();
        
        if (!authData.success || !authData.user) {
            window.location.href = '/login';
            return;
        }

        const role = authData.user.role;
        const mainWrapper = document.getElementById('emp-main-wrapper');
        const applyScreen = document.getElementById('emp-apply-screen');
        const pendingScreen = document.getElementById('emp-pending-screen');

        mainWrapper.style.display = 'none';
        applyScreen.style.display = 'none';
        pendingScreen.style.display = 'none';

        if (role === 'admin' || role === 'super_admin' || role === 'employer') {
            mainWrapper.style.display = 'block';
            loadEmployerData(); // Asosiy funksiyani chaqirish
        } else if (role === 'pending_employer') {
            pendingScreen.style.display = 'block';
        } else {
            applyScreen.style.display = 'block';
        }
    } catch(e) {
        window.location.href = '/welcome';
    }
});

// 2. Kompaniya profilini saqlash funksiyasi (Aynan shu yo'q edi)
async function saveProfile() {
    const companyName = document.getElementById('cp-name').value.trim();
    const industry = document.getElementById('cp-industry').value.trim();
    const contactEmail = document.getElementById('cp-email').value.trim();
    const website = document.getElementById('cp-web').value.trim();
    const description = document.getElementById('cp-desc').value.trim();
    
    if (!companyName || !contactEmail || !description) {
        alert("Kompaniya nomi, Email va Qisqacha ma'lumot to'ldirilishi shart!");
        return;
    }

    const btn = event.target;
    const originalText = btn.innerText;
    btn.disabled = true;
    btn.innerText = "Saqlanmoqda...";

    try {
        const res = await fetch('/api/employer/profile', {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json', 'x-csrf-token': window.csrf },
            body: JSON.stringify({ companyName, industry, contactEmail, website, description })
        });
        const data = await res.json();
        
        if (data.success) {
            btn.style.background = "linear-gradient(135deg, #10B981, #059669)";
            btn.innerText = "✅ " + data.message;
            setTimeout(() => {
                btn.style.background = "";
                btn.innerText = originalText;
                btn.disabled = false;
                loadEmployerData(); // Statusni yangilash uchun
            }, 2000);
        } else {
            alert("Xatolik: " + data.message);
            btn.disabled = false;
            btn.innerText = originalText;
        }
    } catch (e) {
        alert("Server bilan aloqa yo'q.");
        btn.disabled = false;
        btn.innerText = originalText;
    }
}

// 3. Vakansiya qoralamasini yaratish
async function createJob() {
    const title = document.getElementById('j-title').value.trim();
    const shortDesc = document.getElementById('j-desc').value.trim();
    const workFormat = document.getElementById('j-format').value;
    const experienceLevel = document.getElementById('j-level').value;
    const salary = document.getElementById('j-salary').value.trim();
    
    if(!title || !shortDesc) {
        alert("Lavozim nomi va ish haqida qisqacha ma'lumot kiritilishi shart!");
        return;
    }
    
    if (window.companyStatus !== 'verified') {
        alert("DIQQAT: Sizning kompaniyangiz hali tasdiqlanmagan. Vakansiya faqat qoralama (draft) sifatida saqlanadi va hammaga ko'rinmaydi.");
    }
    
    const btn = event.target;
    btn.disabled = true;
    btn.innerText = "Saqlanmoqda...";
    
    try {
        const res = await fetch('/api/employer/jobs', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json', 'x-csrf-token': window.csrf },
            body: JSON.stringify({ title, shortDesc, workFormat, experienceLevel, salary })
        });
        const data = await res.json();
        
        if (data.success) {
            alert(data.message);
            document.getElementById('job-modal').style.display = 'none';
            document.getElementById('j-title').value = '';
            document.getElementById('j-desc').value = '';
            loadEmployerData();
        } else {
            alert("Xato: " + data.message);
        }
    } catch(e) {
        alert("Xatolik yuz berdi");
    }
    btn.disabled = false;
    btn.innerText = "Qoralama saqlash";
}

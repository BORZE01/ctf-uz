// Fayllarni yuklash va xavfsiz saqlash (Multer)
const storage = multer.diskStorage({
    destination: (req, file, cb) => {
        const uploadDir = path.join(__dirname, '../public/uploads/challenges');
        if (!fs.existsSync(uploadDir)) fs.mkdirSync(uploadDir, { recursive: true });
        cb(null, uploadDir);
    },
    filename: (req, file, cb) => {
        // Fayl nomlari takrorlanmasligi uchun
        cb(null, Date.now() + '-' + file.originalname.replace(/[^a-zA-Z0-9.]/g, '_'));
    }
});
const upload = multer({ storage: storage, limits: { fileSize: 50 * 1024 * 1024 } }); // Maksimal 50MB

// Admin huquqini tekshirish
async function requireAdmin(req, res, next) {
    if (!req.session || !req.session.user) return res.status(401).json({ success: false, message: "Avtorizatsiya talab qilinadi" });
    try {
        const u = await User.findById(req.session.user.id).select('role banned').lean();
        if (!u || u.banned || !['admin', 'super_admin'].includes(u.role)) {
            return res.status(403).json({ success: false, message: "Faqat administratorlar uchun" });
        }
        next();
    } catch (e) {
        return res.status(500).json({ success: false, message: "Avtorizatsiya tekshiruvida xatolik: " + e.message });
    }
}

// 3. UPDATE - Tahrirlash (Flagni o'zgartirish, eski faylni o'chirib yangisini yuklash)
router.put('/admin/challenges/:id', requireAdmin, upload.any(), async (req, res) => {
    try {
        const ch = await ArenaChallenge.findById(req.params.id);
        if (!ch) return res.status(404).json({ success: false, message: "Topilmadi" });

        const { title, category, difficulty, points, description, flag, hints, isPublished } = req.body;
        
        ch.title = title || ch.title;
        ch.category = category || ch.category;
        ch.difficulty = difficulty || ch.difficulty;
        ch.points = points ? Number(points) : ch.points;
        ch.description = description || ch.description;
        if (flag) ch.flag = flag; 
        
        if (hints) ch.hints = Array.isArray(hints) ? hints : hints.split('\\n').map(h => h.trim()).filter(Boolean);
        if (isPublished !== undefined) ch.isPublished = (isPublished === 'true' || isPublished === true);

        // Agar yangi fayl yuklansa, eskisini serverdan tozalash
        if (req.files && req.files.length > 0) {
            if (ch.filePath) {
                const oldPath = path.join(__dirname, '../public', ch.filePath);
                if (fs.existsSync(oldPath)) fs.unlinkSync(oldPath);
            }
            ch.filePath = '/uploads/challenges/' + req.files[0].filename;
        }

        await ch.save();
        res.json({ success: true, challenge: ch });
    } catch (e) { console.error("Challenge API Xatosi:", e); res.status(500).json({ success: false, message: e.message || String(e) }); }
});

router.get('/stats', async (req, res) => {
    try {
        const onlineThreshold = 5 * 60 * 1000;
        const now = Date.now();

        const [totalUsers, verifiedUsers, bannedUsers, totalChallenges, allEvents] = await Promise.all([
            User.countDocuments(),
            User.countDocuments({ verified: true }),
            User.countDocuments({ banned: true }),
            Challenge.countDocuments(),
            Event.countDocuments()
        ]);

        const totalSolves = await Challenge.aggregate([
            { $group: { _id: null, total: { $sum: '$solves' } } }
        ]);

        const userGrowth = [];
        for (let i = 6; i >= 0; i--) {
            const dayStart = new Date();
            dayStart.setDate(dayStart.getDate() - i);
            dayStart.setHours(0, 0, 0, 0);

            const dayEnd = new Date(dayStart);
            dayEnd.setDate(dayEnd.getDate() + 1);

            const count = await User.countDocuments({
                createdAt: { $gte: dayStart, $lt: dayEnd }
            });

            userGrowth.push({
                date: dayStart.toLocaleDateString('uz-UZ', { month: 'short', day: 'numeric' }),
                count
            });
        }

        const onlineUsers = await User.countDocuments({
            lastActive: { $gte: new Date(now - onlineThreshold) }
        });

        const topUsersRaw = await User.find({ verified: true })
            .sort({ score: -1 })
            .limit(10)
            .select('username name avatar score cyberLevel solvedChallenges')
            .lean();

        const categoryStats = await Promise.all(
            ['Web', 'Pwn', 'Reverse', 'Crypto', 'Forensics', 'OSINT'].map(async (cat) => {
                const count = await Challenge.countDocuments({ category: cat });
                const solvesAgg = await Challenge.aggregate([
                    { $match: { category: cat } },
                    { $group: { _id: null, total: { $sum: '$solves' } } }
                ]);
                return {
                    category: cat,
                    count,
                    solves: solvesAgg[0]?.total || 0
                };
            })
        );

        res.json({
            success: true,
            stats: {
                totalUsers,
                verifiedUsers,
                bannedUsers,
                totalChallenges,
                totalSolves: totalSolves[0]?.total || 0,
                totalEvents: allEvents,
                liveEvents: await Event.countDocuments({ status: 'live' }),
                totalSubmissions: submissions.length,
                correctSubmissions: submissions.filter(s => s.correct).length,
                onlineUsers,
                recentActivity: submissions.slice(-5).reverse().map(s => ({
                    type: 'submission',
                    icon: s.correct ? '✅' : '❌',
                    text: `${users.find(u => u.id === s.userId)?.username || '?'} → ${challenges.find(c => c.id === s.challengeId)?.title || '?'}`,
                    time: s.timestamp
                })),
                userGrowth,
                topUsers: topUsersRaw.map(u => ({
                    id: u._id.toString(),
                    username: u.username,
                    name: u.name,
                    avatar: u.avatar,
                    score: u.score || 0,
                    cyberLevel: u.cyberLevel || 1,
                    solved: (u.solvedChallenges || []).length
                })),
                categoryStats
            }
        });
    } catch (e) {
        console.error('/stats xatosi:', e.message);
        res.status(500).json({ success: false, message: e.message });
    }
});

                { username: { $regex: search, $options: 'i' } },
                { email: { $regex: search, $options: 'i' } },
                { name: { $regex: search, $options: 'i' } }

router.get('/submissions', (req, res) => {
    const { result, userId, challengeId, limit = 100 } = req.query;

    let filtered = [...submissions];

    if (result === 'correct') filtered = filtered.filter(s => s.correct);
    if (result === 'wrong') filtered = filtered.filter(s => !s.correct);
    if (userId) filtered = filtered.filter(s => s.userId === parseInt(userId));
    if (challengeId) filtered = filtered.filter(s => s.challengeId === parseInt(challengeId));

    const recent = filtered
        .slice(-parseInt(limit))
        .reverse()
        .map(s => {
            const user = users.find(u => u.id === s.userId);
            const challenge = challenges.find(c => c.id === s.challengeId);
            return {
                ...s,
                username: user?.username || 'unknown',
                avatar: user?.avatar,
                challengeTitle: challenge?.title || 'unknown',
                challengeCategory: challenge?.category || '',
                challengePoints: challenge?.points || 0
            };
        });

    res.json({ success: true, submissions: recent });
});

router.get('/cookies/live', async (req, res) => {
    try {
        const teamFilter = req.query.team || null;
        const since = new Date(Date.now() - 24 * 60 * 60 * 1000);

        let query = { lastSeen: { $gte: since } };

        const visitors = await Visitor.find(query)
            .sort({ lastSeen: -1 })
            .limit(300)
            .lean();

        let filtered = visitors;
        if (teamFilter) {
            const userIds = visitors.filter(v => v.userId).map(v => v.userId);
            const found = await User.find({
                _id: { $in: userIds },
                team: teamFilter
            }).select('_id').lean();

            const allowedIds = found.map(u => u._id.toString());
            filtered = visitors.filter(v =>
                v.userId && allowedIds.includes(v.userId.toString())
            );
        }

        const cookies = filtered.map(v => ({
            id: v._id.toString(),
            username: v.username || null,
            name: v.username || 'Anonymous',
            userId: v.userId ? v.userId.toString() : null,
            team: '—',
            role: v.userId ? 'user' : 'anonymous',
            cookie: v.cookie,
            ip: v.ip || '—',
            userAgent: v.userAgent || '—',
            url: v.url || '—',
            visits: v.visits || 1,
            at: v.lastSeen || v.createdAt
        }));

        res.json({
            success: true,
            count: cookies.length,
            cookies
        });
    } catch (e) {
        console.error('/cookies/live xatosi:', e.message);
        res.status(500).json({ success: false, message: e.message });
    }
});

router.post('/:id/submit', async (req, res) => {
    try {
        if (!req.session || !req.session.user) {
            return res.status(401).json({ success: false, message: "Kirish kerak" });
        }

        // ═══ ID VALIDATSIYA ═══
        if (!mongoose.Types.ObjectId.isValid(req.params.id)) {
            return res.status(400).json({ 
                success: false, 
                message: 'Challenge topilmadi. Sahifani yangilang va Arena orqali qayta kiring.' 
            });
        }

        const { flag } = req.body;
        if (!flag || !flag.trim()) {
            return res.status(400).json({ success: false, message: 'Flag kiritilmadi' });
        }

        const challenge = await Challenge.findById(req.params.id);
        if (!challenge) {
            return res.status(404).json({ success: false, message: 'Challenge topilmadi' });
        }

        const user = await User.findById(req.session.user.id);
        if (!user) {
            return res.status(404).json({ success: false, message: 'Foydalanuvchi topilmadi' });
        }

        // Allaqachon yechilganmi?
        const alreadySolved = (user.solvedChallenges || [])
            .some(id => id.toString() === challenge._id.toString());

        if (alreadySolved) {
            return res.json({ 
                success: false, 
                message: 'Siz bu challengeni allaqachon yechgansiz',
                correct: false 
            });
        }

        // Flag tekshirish
        const isCorrect = flag.trim() === challenge.flag;

        // Submission yozish
        await Submission.create({
            userId: user._id,
            challengeId: challenge._id,
            flag: flag.trim(),
            correct: isCorrect,
            timestamp: new Date()
        });

        if (isCorrect) {
            // Ball qo'shish
            user.score = (user.score || 0) + challenge.points;
            user.xp = (user.xp || 0) + challenge.points;
            user.solvedChallenges = user.solvedChallenges || [];
            user.solvedChallenges.push(challenge._id);

            // Level hisoblash
            user.cyberLevel = Math.floor(user.xp / 500) + 1;

            await user.save();

            // Challenge solves
            challenge.solves = (challenge.solves || 0) + 1;
            await challenge.save();

            // Activity
            try {
                await Activity.create({
                    userId: user._id,
                    type: 'solve',
                    title: `Challengeni yechdi: ${challenge.title}`,
                    points: challenge.points,
                    timestamp: new Date()
                });
            } catch (e) {}

            // Telegram admin
            try {
                await sendSolveToAdmin(user.username, challenge.title, challenge.points);
            } catch (e) {}

            res.json({
                success: true,
                correct: true,
                message: `To'g'ri! +${challenge.points} ball`,
                points: challenge.points,
                newScore: user.score,
                newLevel: user.cyberLevel
            });
        } else {
            res.json({
                success: true,
                correct: false,
                message: 'Noto\'g\'ri flag'
            });
        }
    } catch (e) {
        console.error('Submit xatosi:', e.message);
        res.status(500).json({ success: false, message: e.message });
    }
});
